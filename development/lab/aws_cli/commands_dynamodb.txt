1. Crear la tabla productos (clave: sku)

AWS_ACCESS_KEY_ID=test \
AWS_SECRET_ACCESS_KEY=test \
AWS_DEFAULT_REGION=us-east-1 \
aws --endpoint-url=http://localhost:4566 \
    dynamodb create-table \
    --table-name productos \
    --attribute-definitions AttributeName=sku,AttributeType=S \
    --key-schema AttributeName=sku,KeyType=HASH \
    --billing-mode PAY_PER_REQUEST



2. Guardar productos reales (put-item)

AWS_ACCESS_KEY_ID=test \
AWS_SECRET_ACCESS_KEY=test \
AWS_DEFAULT_REGION=us-east-1 \
aws --endpoint-url=http://localhost:4566 \
    dynamodb put-item \
    --table-name productos \
    --item '{
        "sku": {"S": "LAP-01"},
        "nombre": {"S": "Laptop Lenovo ThinkPad"},
        "precio": {"N": "16500"},
        "categoria": {"S": "Computo"},
        "stock": {"N": "8"}
    }'

3. Buscar un producto por su código (get-item)

AWS_ACCESS_KEY_ID=test \
AWS_SECRET_ACCESS_KEY=test \
AWS_DEFAULT_REGION=us-east-1 \
aws --endpoint-url=http://localhost:4566 \
    dynamodb get-item \
    --table-name productos \
    --key '{"sku": {"S": "LAP-01"}}'


4. Consultar todo el inventario (scan)

AWS_ACCESS_KEY_ID=test \
AWS_SECRET_ACCESS_KEY=test \
AWS_DEFAULT_REGION=us-east-1 \
aws --endpoint-url=http://localhost:4566 \
    dynamodb scan \
    --table-name productos


5. Actualizar el precio y stock tras una venta (update-item)


AWS_ACCESS_KEY_ID=test \
AWS_SECRET_ACCESS_KEY=test \
AWS_DEFAULT_REGION=us-east-1 \
aws --endpoint-url=http://localhost:4566 \
    dynamodb update-item \
    --table-name productos \
    --key '{"sku": {"S": "LAP-01"}}' \
    --update-expression "SET precio = :nuevo_precio, stock = :nuevo_stock" \
    --expression-attribute-values '{
        ":nuevo_precio": {"N": "15999"},
        ":nuevo_stock": {"N": "7"}
    }'
